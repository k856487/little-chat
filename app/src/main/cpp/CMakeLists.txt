cmake_minimum_required(VERSION 3.22.1)
project(littlechat)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp/CMakeLists.txt")
  message(FATAL_ERROR "llama.cpp source missing. Run the GitHub Actions build workflow or clone ggml-org/llama.cpp into app/src/main/cpp/llama.cpp.")
endif()
add_subdirectory(llama.cpp)
add_library(littlechat SHARED native-lib.cpp)
find_library(log-lib log)
target_link_libraries(littlechat PRIVATE llama ${log-lib})
